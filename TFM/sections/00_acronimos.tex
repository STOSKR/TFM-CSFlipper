\chapter*{Lista de acrónimos}
\addcontentsline{toc}{chapter}{Lista de acrónimos}
\indicetoprule

\begin{description}
  \item[\acrotarget{API}{API}] Interfaz de programación de aplicaciones.
  \item[\acrotarget{BUFF}{BUFF}] Mercado externo de compraventa de activos digitales usado como fuente de precios.
  \item[\acrotarget{CLI}{CLI}] Interfaz de línea de comandos, del inglés Command-Line Interface.
  \item[\acrotarget{CNY}{CNY}] Yuan chino, código internacional de divisa.
  \item[\acrotarget{CS2}{CS2}] Counter-Strike 2.
  \item[\acrotarget{CSV}{CSV}] Valores separados por comas, del inglés Comma-Separated Values.
  \item[\acrotarget{CTDE}{CTDE}] Entrenamiento centralizado y ejecución descentralizada, del inglés Centralized Training with Decentralized Execution.
  \item[\acrotarget{EUR}{EUR}] Euro, código internacional de divisa.
  \item[\acrotarget{F1}{F1}] Métrica F1, media armónica entre precisión y exhaustividad.
  \item[\acrotarget{IA}{IA}] Inteligencia artificial.
  \item[\acrotarget{MADDPG}{MADDPG}] Gradiente de política determinista profundo multiagente, del inglés Multi-Agent Deep Deterministic Policy Gradient.
  \item[\acrotarget{MAPPO}{MAPPO}] Optimización proximal de políticas multiagente, del inglés Multi-Agent Proximal Policy Optimization.
  \item[\acrotarget{MARL}{MARL}] Aprendizaje por refuerzo multiagente, del inglés Multi-Agent Reinforcement Learning.
  \item[\acrotarget{MDP}{MDP}] Proceso de decisión de Markov, del inglés Markov Decision Process.
  \item[\acrotarget{OCR}{OCR}] Reconocimiento óptico de caracteres, del inglés Optical Character Recognition.
  \item[\acrotarget{CLAHE}{CLAHE}] Ecualización adaptativa de histograma con límite de contraste, del inglés Contrast Limited Adaptive Histogram Equalization.
  \item[\acrotarget{PPO}{PPO}] Proximal Policy Optimization.
  \item[\acrotarget{PR-AUC}{PR-AUC}] Área bajo la curva precisión-exhaustividad, del inglés Precision-Recall Area Under the Curve.
  \item[\acrotarget{RL}{RL}] Aprendizaje por refuerzo, del inglés Reinforcement Learning.
  \item[\acrotarget{RLlib}{RLlib}] Biblioteca de aprendizaje por refuerzo de Ray.
  \item[\acrotarget{ROC-AUC}{ROC-AUC}] Área bajo la curva ROC, del inglés Receiver Operating Characteristic Area Under the Curve.
  \item[\acrotarget{RSI}{RSI}] Índice de fuerza relativa, del inglés Relative Strength Index.
  \item[\acrotarget{TFM}{TFM}] Trabajo Fin de Máster.
\end{description}
